Category 1

Database Health Check Scripts

Run these daily to confirm your database is healthy, components are valid, and no hidden issues exist.

Safe

Database Status & Mode

SELECT name, db_unique_name, open_mode,
       log_mode, protection_mode,
       database_role, flashback_on,
       TO_CHAR(created,'YYYY-MM-DD') created
FROM   v$database;
Safe

Invalid DB Components

SELECT comp_id, comp_name, version, status
FROM   dba_registry
WHERE  status != 'VALID'
ORDER BY comp_id;
Safe

Invalid Objects (All Users)

SELECT owner, object_type,
       COUNT(*) invalid_count
FROM   dba_objects
WHERE  status = 'INVALID'
AND    owner NOT IN ('SYS','SYSTEM')
GROUP BY owner, object_type
ORDER BY invalid_count DESC;
Safe

Recent ORA- Errors from Alert Log

-- 19c+ External Table view of alert log
SELECT originating_timestamp, message_text
FROM   x$dbgalertext
WHERE  message_text LIKE 'ORA-%'
AND    originating_timestamp > SYSTIMESTAMP-1
ORDER BY originating_timestamp DESC;
Category 2

Space & Tablespace Reports

Monitor disk usage before it becomes a problem. These scripts help prevent ORA-01653 (out of space) emergencies.

Safe

Tablespace Usage Summary

SELECT
  tablespace_name,
  ROUND(used_space*8192/1024/1024/1024,2) used_gb,
  ROUND(tablespace_size*8192/1024/1024/1024,2) total_gb,
  ROUND(used_percent,1) pct_used,
  CASE WHEN used_percent > 90 THEN 'CRITICAL'
       WHEN used_percent > 80 THEN 'WARNING'
       ELSE 'OK' END status
FROM   dba_tablespace_usage_metrics
ORDER BY used_percent DESC;
Safe

Top Space Consumers by Segment

SELECT owner, segment_name, segment_type,
       ROUND(bytes/1024/1024/1024,2) size_gb
FROM   dba_segments
WHERE  owner NOT IN ('SYS','SYSTEM','SYSAUX')
ORDER BY bytes DESC
FETCH FIRST 20 ROWS ONLY;
Safe

FRA (Flash Recovery Area) Usage

SELECT
  name,
  ROUND(space_limit/1024/1024/1024,1) limit_gb,
  ROUND(space_used/1024/1024/1024,1) used_gb,
  ROUND(space_used*100/space_limit,1) pct_used
FROM   v$recovery_file_dest;
Safe

Datafiles & Autoextend Status

SELECT tablespace_name,
       file_name,
       ROUND(bytes/1024/1024,1) size_mb,
       autoextensible,
       ROUND(maxbytes/1024/1024/1024,1) max_gb
FROM   dba_data_files
ORDER BY tablespace_name, file_name;
Category 3

Session & Locking Scripts

Identify blocking sessions, find long-running queries, and monitor active connections.

Safe

Blocking Sessions

SELECT
  s.blocking_session blocker,
  s.sid victim, s.serial#,
  s.username, s.machine,
  s.wait_class, s.event,
  s.seconds_in_wait secs,
  sq.sql_text
FROM   v$session s
LEFT JOIN v$sql sq ON s.sql_id = sq.sql_id
WHERE  s.blocking_session IS NOT NULL
ORDER BY s.seconds_in_wait DESC;
Safe

Long-Running Operations

SELECT
  sid, serial#, opname, target,
  sofar, totalwork,
  ROUND(100*sofar/DECODE(totalwork,0,1,totalwork),1) pct,
  time_remaining secs_left,
  elapsed_seconds
FROM   v$session_longops
WHERE  time_remaining > 0
ORDER BY elapsed_seconds DESC;
Safe

Active Sessions by Wait Class

SELECT
  wait_class,
  COUNT(*) session_count,
  ROUND(AVG(seconds_in_wait),1) avg_wait_secs
FROM   v$session
WHERE  status = 'ACTIVE'
AND    type = 'USER'
GROUP BY wait_class
ORDER BY session_count DESC;
Caution

Kill a Session

-- First: identify the session
SELECT sid, serial#, username, status, sql_id
FROM   v$session
WHERE  username = 'APP_USER';

-- Then: kill it (replace SID,SERIAL#)
ALTER SYSTEM KILL SESSION '123,456' IMMEDIATE;

-- For OS-level kill (if ORA-00031):
-- Get SPID from v$process, then OS kill -9 $SPID
Category 4

Performance Monitoring Scripts

Find high-resource SQL, check wait events, and identify performance bottlenecks in real time.

Safe

Top SQL by CPU

SELECT sql_id,
       ROUND(cpu_time/1e6/DECODE(executions,0,1,executions),2) avg_cpu_sec,
       executions,
       ROUND(buffer_gets/DECODE(executions,0,1,executions)) avg_lio,
       SUBSTR(sql_text,1,60) sql_preview
FROM   v$sql
WHERE  executions > 0
ORDER BY cpu_time DESC
FETCH FIRST 10 ROWS ONLY;
Safe

System-Wide Wait Events

SELECT event, wait_class,
       total_waits,
       ROUND(time_waited/100,1) time_waited_sec,
       ROUND(average_wait/100,4) avg_wait_sec
FROM   v$system_event
WHERE  wait_class != 'Idle'
ORDER BY time_waited DESC
FETCH FIRST 15 ROWS ONLY;
Safe

Buffer Cache Hit Ratio

SELECT
  ROUND((1 - (phy.value/(log.value+con.value)))*100,2) hit_ratio_pct
FROM
  (SELECT value FROM v$sysstat WHERE name='physical reads') phy,
  (SELECT value FROM v$sysstat WHERE name='db block gets') log,
  (SELECT value FROM v$sysstat WHERE name='consistent gets') con;
-- Target: > 95% (OLTP), > 80% (DW)
Safe

SGA Component Sizes

SELECT
  component,
  ROUND(current_size/1024/1024,0) current_mb,
  ROUND(min_size/1024/1024,0) min_mb,
  ROUND(max_size/1024/1024,0) max_mb
FROM   v$sga_dynamic_components
WHERE  current_size > 0
ORDER BY current_size DESC;
Category 5

Backup & Recovery Status Scripts

Verify your backups succeeded, check archivelog status, and confirm recovery readiness.

Safe

RMAN Backup Status (Last 7 Days)

SELECT
  session_key,
  input_type,
  status,
  TO_CHAR(start_time,'YYYY-MM-DD HH24:MI') started,
  ROUND(elapsed_seconds/60,1) elapsed_min,
  ROUND(output_bytes/1024/1024/1024,2) output_gb
FROM   v$rman_backup_job_details
WHERE  start_time > SYSDATE-7
ORDER BY start_time DESC;
Safe

Archivelog Generation Rate

SELECT
  TRUNC(first_time,'HH') hour_bucket,
  COUNT(*) archivelogs,
  ROUND(SUM(blocks*block_size)/1024/1024,0) mb_generated
FROM   v$archived_log
WHERE  first_time > SYSDATE-1
AND    standby_dest = 'NO'
GROUP BY TRUNC(first_time,'HH')
ORDER BY 1;
Category 6

Security Audit Scripts

Check for security misconfigurations, excessive privileges, and default passwords.

Safe

Users with DBA Role

SELECT grantee, granted_role,
       admin_option, default_role
FROM   dba_role_privs
WHERE  granted_role = 'DBA'
AND    grantee NOT IN ('SYS','SYSTEM')
ORDER BY grantee;
Important

Default Passwords Check

SELECT username, account_status,
       expiry_date, default_tablespace
FROM   dba_users
WHERE  password_versions LIKE '%10G%'
OR     username IN (
  SELECT username FROM dba_users_with_defpwd
)
ORDER BY username;
Category 7

Object Management Scripts

Find large tables, check index status, identify fragmented objects, and analyze object growth.

Safe

Tables Without Primary Key

SELECT owner, table_name
FROM   dba_tables
WHERE  owner NOT IN ('SYS','SYSTEM','SYSAUX')
MINUS
SELECT owner, table_name
FROM   dba_constraints
WHERE  constraint_type = 'P'
ORDER BY 1,2;
Safe

Unusable Indexes

SELECT owner, index_name, table_name,
       status, partitioned
FROM   dba_indexes
WHERE  status = 'UNUSABLE'
AND    owner NOT IN ('SYS','SYSTEM')
ORDER BY owner, table_name;
