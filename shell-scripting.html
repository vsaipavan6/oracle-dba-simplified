#!/bin/bash
# Quick tablespace check from shell
. /home/oracle/.bash_profile   # Load Oracle env

sqlplus -s / as sysdba <<EOF
SET LINESIZE 120 PAGESIZE 50 FEEDBACK OFF
SELECT tablespace_name,
       ROUND(used_space*8192/1024/1024,1) used_mb,
       ROUND(tablespace_size*8192/1024/1024,1) total_mb,
       ROUND(100*used_percent,1) pct_used
FROM   dba_tablespace_usage_metrics
ORDER BY pct_used DESC;
EOF

Script Structure Template

#!/bin/bash
# Script: rman_backup.sh
# Purpose: Daily RMAN incremental backup
# Author: DBA Team | Date: 2025-01
# Usage: ./rman_backup.sh [0|1]

# === STRICT MODE ===
set -euo pipefail
IFS=$'\n\t'

# === ENVIRONMENT ===
export ORACLE_SID=ORCL
export ORACLE_HOME=/u01/app/oracle/product/19c/dbhome_1
export PATH=$ORACLE_HOME/bin:$PATH
LOG_DIR=/u01/logs/rman
DATE=$(date +%Y%m%d_%H%M%S)
LOGFILE=$LOG_DIR/rman_$DATE.log
LEVEL=${1:-1}   # Default level 1

# === FUNCTIONS ===
log() { echo "[$(date '+%H:%M:%S')] $*" | tee -a $LOGFILE; }
die() { log "ERROR: $*"; exit 1; }

# === MAIN ===
mkdir -p $LOG_DIR
log "Starting Level $LEVEL backup for $ORACLE_SID"

Conditionals, Loops & Functions

# === CONDITIONAL ===
if [[ $LEVEL -eq 0 ]]; then
  log "Running Level 0 (full)"
elif [[ $LEVEL -eq 1 ]]; then
  log "Running Level 1 (incremental)"
else
  die "Invalid level: $LEVEL"
fi

# === LOOP: process multiple SIDs ===
for SID in ORCL DEVDB TESTDB; do
  export ORACLE_SID=$SID
  log "Backing up $SID..."
  run_backup
done

# === CHECK EXIT CODE ===
if [[ $? -ne 0 ]]; then
  send_alert "Backup FAILED for $ORACLE_SID"
  exit 1
fi

# === READ FILE LINE BY LINE ===
while IFS= read -r line; do
  echo "Processing: $line"
done < /u01/config/db_list.txt
Section 2

SQL*Plus in Shell Scripts

Embed SQL queries in shell scripts to capture output, check thresholds, and act on results automatically.

Capture SQL Output into Shell Variable

#!/bin/bash
. /home/oracle/.bash_profile

# Capture a value from SQL query
DB_STATUS=$(sqlplus -s / as sysdba <<EOF
SET HEADING OFF FEEDBACK OFF PAGESIZE 0
SELECT open_mode FROM v\$database;
EOF)

DB_STATUS=$(echo $DB_STATUS | tr -d ' ')

if [[ $DB_STATUS == "READWRITE" ]]; then
  echo "Database is OPEN and READ WRITE — OK"
else
  echo "WARNING: DB status is $DB_STATUS"
  send_alert "$ORACLE_SID not in READ WRITE: $DB_STATUS"
fi

Run SQL File & Check Exit Code

#!/bin/bash
# Run a SQL script and validate success
sqlplus -s / as sysdba <<EOF
  WHENEVER SQLERROR EXIT 1
  WHENEVER OSERROR EXIT 2
  @/u01/scripts/create_indexes.sql
  COMMIT;
  EXIT 0;
EOF

RC=$?
if [[ $RC -ne 0 ]]; then
  log "SQL script failed with RC=$RC"
  send_alert "Index creation FAILED on $ORACLE_SID"
  exit 1
fi
log "SQL script completed successfully"

RMAN Backup Automation Script

A production-ready RMAN backup shell script with logging, alerting, and retention cleanup.

rman_backup.sh — Production Ready 
#!/bin/bash
# rman_backup.sh — Run Level 0 or Level 1 RMAN backup
set -euo pipefail

export ORACLE_SID=ORCL
export ORACLE_HOME=/u01/app/oracle/product/19c/dbhome_1
export PATH=$ORACLE_HOME/bin:$PATH
LOG=/u01/logs/rman/rman_$(date +%Y%m%d).log
LEVEL=${1:-1}
EMAIL=dba-alerts@company.com

log() { echo "[$(date '+%F %T')] $*" | tee -a $LOG; }

send_alert() {
  echo "$1" | mail -s "[RMAN ALERT] $ORACLE_SID" $EMAIL
}

mkdir -p /u01/logs/rman
log "=== RMAN Backup Start: Level $LEVEL ==="

rman target / log=$LOG append <<EOF
RUN {
  ALLOCATE CHANNEL c1 DEVICE TYPE DISK FORMAT '/u02/backup/%U';
  ALLOCATE CHANNEL c2 DEVICE TYPE DISK FORMAT '/u02/backup/%U';
  BACKUP INCREMENTAL LEVEL $LEVEL
    DATABASE PLUS ARCHIVELOG DELETE INPUT;
  DELETE NOPROMPT OBSOLETE;
}
EOF

RC=$?
if [[ $RC -ne 0 ]]; then
  log "RMAN backup FAILED (RC=$RC)"
  send_alert "RMAN backup FAILED on $ORACLE_SID. Check $LOG"
  exit 1
fi

log "=== RMAN Backup Completed Successfully ==="

Daily Health Check Script

Run every morning — database status, tablespace usage, failed jobs, blocking sessions, and archivelog space.

db_health_check.sh 
#!/bin/bash
. /home/oracle/.bash_profile
REPORT=/tmp/health_$ORACLE_SID_$(date +%Y%m%d).txt
exec > $REPORT 2>&1

echo "========================================="
echo "  Oracle DB Health Check — $ORACLE_SID"
echo "  $(date '+%F %T')"
echo "========================================="

sqlplus -s / as sysdba <<EOF
SET LINESIZE 130 PAGESIZE 60 FEEDBACK OFF

PROMPT --- DATABASE STATUS ---
SELECT name, db_unique_name, open_mode, log_mode, created
FROM   v\$database;

PROMPT --- TABLESPACE USAGE (>70%) ---
SELECT tablespace_name,
       ROUND(used_percent,1)||'%' pct_used
FROM   dba_tablespace_usage_metrics
WHERE  used_percent > 70
ORDER BY used_percent DESC;

PROMPT --- FAILED SCHEDULER JOBS (last 24h) ---
SELECT job_name, status, run_duration, error#
FROM   dba_scheduler_job_run_details
WHERE  status != 'SUCCEEDED'
AND    actual_start_date > SYSDATE-1;

PROMPT --- BLOCKING SESSIONS ---
SELECT blocking_session, sid, wait_class,
       seconds_in_wait, event
FROM   v\$session
WHERE  blocking_session IS NOT NULL;

PROMPT --- FRA USAGE ---
SELECT ROUND(space_used/space_limit*100,1)||'%' fra_pct_used
FROM   v\$recovery_file_dest;
EOF

# Email the report
mail -s "[$ORACLE_SID] Health Check $(date +%F)" dba@company.com < $REPORT

Cron Scheduling for DBAs

Schedule DBA scripts with cron. Always redirect output to log files — cron has no terminal and emails stderr by default.

Cron Syntax Reference

# ┌──── minute (0-59)
# │ ┌─── hour (0-23)
# │ │ ┌── day-of-month (1-31)
# │ │ │ ┌─ month (1-12)
# │ │ │ │ ┌ day-of-week (0=Sun)
# │ │ │ │ │
# * * * * *  command

# Examples:
0  2  * * 0   /dba/scripts/rman_backup.sh 0   # Level 0 Sunday 2am
0  2  * * 1-6 /dba/scripts/rman_backup.sh 1   # Level 1 Mon-Sat 2am
0  6  * * *   /dba/scripts/health_check.sh    # Daily health check 6am
*/15 * * * *  /dba/scripts/tbs_alert.sh       # Tablespace alert every 15m
30 23  * * 5   /dba/scripts/weekly_report.sh   # Weekly report Friday 11:30pm

Setting Up Oracle Cron

# Edit oracle user's crontab
crontab -e -u oracle

# CRITICAL: Set ORACLE env in crontab itself
# Cron doesn't source .bash_profile!
ORACLE_SID=ORCL
ORACLE_HOME=/u01/app/oracle/product/19c/dbhome_1
PATH=$ORACLE_HOME/bin:/usr/bin:/bin
MAILTO=oracle@company.com

0 2 * * * /dba/scripts/rman_backup.sh 1 >> /u01/logs/cron.log 2&1

# List current cron jobs
crontab -l -u oracle

# Check cron execution in syslog
grep CRON /var/log/cron | tail -20

Tablespace Alert Monitoring Script

Real-time alerting when tablespace crosses threshold. Run every 15 minutes via cron.

tbs_alert.sh — Tablespace Threshold Alerting 
#!/bin/bash
. /home/oracle/.bash_profile
THRESHOLD=85
EMAIL=dba-alerts@company.com

ALERTS=$(sqlplus -s / as sysdba <<EOF
SET HEADING OFF FEEDBACK OFF PAGESIZE 0 LINESIZE 200
SELECT tablespace_name||'|'||ROUND(used_percent,1)
FROM   dba_tablespace_usage_metrics
WHERE  used_percent > $THRESHOLD;
EOF)

if [[ -n "$ALERTS" ]]; then
  MSG="TABLESPACE ALERT on $ORACLE_SID:\n\nTablespace|Pct_Used\n$ALERTS\n\nAction: Add datafile or extend existing."
  echo -e "$MSG" | mail -s "[P2 ALERT] Tablespace > ${THRESHOLD}%" $EMAIL
  logger -t ORA_TBS_ALERT "Tablespace alert: $ALERTS"
fi

Shell Scripting — Interview Q&A

Cron runs with a minimal environment — it doesn't source .bash_profile or .bashrc. So ORACLE_HOME, ORACLE_SID, and PATH are not set. Fix: Either source the profile explicitly (. /home/oracle/.bash_profile) at the top of your script, or set the variables directly in the crontab file before the job entries.

Use a here-document with sqlplus -s (silent mode) and command substitution: VAR=$(sqlplus -s / as sysdba <<EOF ... EOF). Set SET HEADING OFF FEEDBACK OFF PAGESIZE 0 in the SQL block so only the data value is returned. Then strip whitespace with tr -d ' ' or xargs before comparing.

  • -e — Exit immediately if any command returns non-zero
  • -u — Treat unset variables as errors (prevents subtle bugs)
  • -o pipefail — Pipeline returns failure if any command in it fails (not just the last)
  • Together they make scripts fail loudly and fast rather than silently continuing after errors — critical for production DBA automation.
#!/bin/bash
# Monitor alert log for ORA- errors
ALERT_LOG=$(find $ORACLE_BASE/diag -name "alert_${ORACLE_SID}.log" 2/dev/null)
tail -0f $ALERT_LOG | while read line; do
  if echo "$line" | grep -qE "ORA-[0-9]+"; then
    echo "$line" | mail -s "ORA- error on $ORACLE_SID" $EMAIL
  fi
done
